# TFR_中俄谈判 - 边境冲突脚本效果
# 黑线 SOV（非梅德韦杰夫/非红色）时替代谈判路线
# 逐州边境战争（原生 start_border_war），按相邻可达推进
# 双轨优势：PRC.adv_* 变量 → 目标州 dynamic modifier（local_attack/defense 变量绑定）
# 全面升级：365 天限时任务，时间耗尽判定中国输（白和+赔款）
# 库页岛(537)：其余 7 州全部收回后自动获得
# 运行作用域：全部在 PRC 作用域调用（除注明外）

##############################################################################
# 冲突初始化 — 启动边境冲突时调用
##############################################################################
TFR_CRN_conflict_init = {
	set_global_flag = TFR_CRN_conflict_active
	set_global_flag = TFR_CRN_conflict_accessible
	PRC = {
		set_country_flag = TFR_CRN_PRC_started_conflict
		set_variable = { var = TFR_CRN_prc_bonus value = 0 }
		set_variable = { var = TFR_CRN_sov_bonus value = 0 }
		set_variable = { var = TFR_CRN_prc_advantage value = 0 }
		set_variable = { var = TFR_CRN_sov_advantage value = 0 }
		set_variable = { var = TFR_CRN_states_taken value = 0 }
	}
	TFR_CRN_clear_target_flags = yes
	TFR_CRN_calc_advantage = yes
	TFR_CRN_advance_target = yes
	PRC = { country_event = { id = TFR_CRN.20 } }
}

# 清除全部目标州旗标
##############################################################################

##############################################################################
# 优势重算 — 基础 0 + 形势加成（SOV 战争/投降倾向/稳定度）+ 施压累计（上限 100）
# 施压累计满 100 → 对方忍无可忍直接宣战（TFR_CRN_check_pressure_explosion）
# 运行在 PRC 作用域
# 注意：SOV 情景加成在 SOV 域计算后写 SOV 持久变量，再回 PRC 组装——
#        temp 变量存在当前作用域对象上，跨作用域累加会丢失（历史 bug 已修）
##############################################################################
TFR_CRN_calc_advantage = {
	set_temp_variable = { adv_prc = SOV.TFR_CRN_sov_situational }
	add_to_temp_variable = { adv_prc = TFR_CRN_prc_bonus }
	set_temp_variable = { adv_sov = 0 }
	add_to_temp_variable = { adv_sov = TFR_CRN_sov_bonus }
	# 钳位 0~100
	if = { limit = { check_variable = { adv_prc > 100 } } set_temp_variable = { adv_prc = 100 } }
	if = { limit = { check_variable = { adv_prc < 0 } } set_temp_variable = { adv_prc = 0 } }
	if = { limit = { check_variable = { adv_sov > 100 } } set_temp_variable = { adv_sov = 100 } }
	if = { limit = { check_variable = { adv_sov < 0 } } set_temp_variable = { adv_sov = 0 } }
	set_variable = { var = TFR_CRN_prc_advantage value = adv_prc }
	set_variable = { var = TFR_CRN_sov_advantage value = adv_sov }
	TFR_CRN_update_advantage_frames = yes
	TFR_CRN_check_pressure_explosion = yes
}

##############################################################################
# SOV 情景加成计算 — 运行在 SOV 作用域
# 结果写 SOV.TFR_CRN_sov_situational（0~45），由 calc_advantage 在 PRC 域读取
# 问题修复：原实现把这三项累加进 PRC 域的 temp 变量，在 SOV 子块内实际落到了
#           SOV 域，导致战争形势加成全部丢失（数值崩塌）；现改为持久变量中转
##############################################################################
TFR_CRN_calc_sov_situational = {
	set_variable = { var = TFR_CRN_sov_situational value = 0 }
	# SOV 深陷战争 → +15
	if = { limit = { has_war = yes } add_to_variable = { var = TFR_CRN_sov_situational value = 15 } }
	# 投降倾向（0~100% → 0~30）
	set_temp_variable = { surv = surrender_progress }
	multiply_temp_variable = { surv = 0.3 }
	round_variable = surv
	add_to_variable = { var = TFR_CRN_sov_situational value = surv }
	# 稳定度惩罚（低于 100% 每 1% 扣 1，最多 15）
	set_temp_variable = { stab = stability }
	multiply_temp_variable = { stab = 100 }
	subtract_from_temp_variable = { stab = 100 }
	multiply_temp_variable = { stab = -1 }
	if = { limit = { check_variable = { stab < 0 } } set_temp_variable = { stab = 0 } }
	if = { limit = { check_variable = { stab > 15 } } set_temp_variable = { stab = 15 } }
	round_variable = stab
	add_to_variable = { var = TFR_CRN_sov_situational value = stab }
	# 钳位 0~45
	if = { limit = { check_variable = { TFR_CRN_sov_situational > 45 } } set_variable = { var = TFR_CRN_sov_situational value = 45 } }
	if = { limit = { check_variable = { TFR_CRN_sov_situational < 0 } } set_variable = { var = TFR_CRN_sov_situational value = 0 } }
}

##############################################################################
# 施压引爆检查 — 任一方施压累计满 100 → 对方忍无可忍直接宣战
# 中方施压满 → 俄方宣战中国；俄方施压满 → 中国宣战俄方
# 运行在 PRC 作用域（每个施压决议后由 calc_advantage 调用）
##############################################################################
TFR_CRN_check_pressure_explosion = {
	if = {
		limit = {
			NOT = { has_global_flag = TFR_CRN_full_war_active }
			NOT = { has_global_flag = TFR_CRN_conflict_ended }
			has_global_flag = TFR_CRN_conflict_active
			NOT = { check_variable = { TFR_CRN_prc_bonus < 100 } }
		}
		TFR_CRN_pressure_explosion_by_sov = yes
	}
	else_if = {
		limit = {
			NOT = { has_global_flag = TFR_CRN_full_war_active }
			NOT = { has_global_flag = TFR_CRN_conflict_ended }
			has_global_flag = TFR_CRN_conflict_active
			NOT = { check_variable = { TFR_CRN_sov_bonus < 100 } }
		}
		TFR_CRN_pressure_explosion_by_prc = yes
	}
}

##############################################################################
# 进度条帧计算（0~100 → 帧 1~21）— 优势双条 + 施压双条
##############################################################################
TFR_CRN_update_advantage_frames = {
	set_variable = { var = TFR_CRN_prc_adv_frame value = TFR_CRN_prc_advantage }
	multiply_variable = { var = TFR_CRN_prc_adv_frame value = 0.2 }
	round_variable = TFR_CRN_prc_adv_frame
	add_to_variable = { TFR_CRN_prc_adv_frame = 1 }
	set_variable = { var = TFR_CRN_sov_adv_frame value = TFR_CRN_sov_advantage }
	multiply_variable = { var = TFR_CRN_sov_adv_frame value = 0.2 }
	round_variable = TFR_CRN_sov_adv_frame
	add_to_variable = { TFR_CRN_sov_adv_frame = 1 }
	set_variable = { var = TFR_CRN_prc_pressure_frame value = TFR_CRN_prc_bonus }
	multiply_variable = { var = TFR_CRN_prc_pressure_frame value = 0.2 }
	round_variable = TFR_CRN_prc_pressure_frame
	add_to_variable = { TFR_CRN_prc_pressure_frame = 1 }
	set_variable = { var = TFR_CRN_sov_pressure_frame value = TFR_CRN_sov_bonus }
	multiply_variable = { var = TFR_CRN_sov_pressure_frame value = 0.2 }
	round_variable = TFR_CRN_sov_pressure_frame
	add_to_variable = { TFR_CRN_sov_pressure_frame = 1 }
}

##############################################################################
# 目标州挂载州级战斗修正（五段制）— 运行在 PRC
# 按优势差 D = PRC优势 - SOV优势 选档（t0 不挂 / t1~t5）：
#   D ≥80 → t5(+25%)   ≥60 → t4(+20%)   ≥40 → t3(+15%)   ≥20 → t2(+10%)   ≥0 → t1(+5%)   D<0 → 不挂
# 档位写入 PRC.TFR_CRN_current_tier，由 STATE 作用域 helper 执行挂载
##############################################################################
TFR_CRN_attach_advantage_modifier = {
	set_temp_variable = { adv_diff = TFR_CRN_prc_advantage }
	subtract_from_temp_variable = { adv_diff = TFR_CRN_sov_advantage }
	set_temp_variable = { tier = 0 }
	if = { limit = { NOT = { check_variable = { adv_diff < 80 } } } set_temp_variable = { tier = 5 } }
	else_if = { limit = { NOT = { check_variable = { adv_diff < 60 } } } set_temp_variable = { tier = 4 } }
	else_if = { limit = { NOT = { check_variable = { adv_diff < 40 } } } set_temp_variable = { tier = 3 } }
	else_if = { limit = { NOT = { check_variable = { adv_diff < 20 } } } set_temp_variable = { tier = 2 } }
	else_if = { limit = { NOT = { check_variable = { adv_diff < 0 } } } set_temp_variable = { tier = 1 } }
	else = { set_temp_variable = { tier = 0 } }
	set_variable = { var = TFR_CRN_current_tier value = tier }
	# 对当前目标州挂载（tier=0 时 helper 不挂任何东西）
	if = { limit = { has_global_flag = TFR_CRN_target_408 } 408 = { TFR_CRN_attach_tier_state = yes } }
	if = { limit = { has_global_flag = TFR_CRN_target_409 } 409 = { TFR_CRN_attach_tier_state = yes } }
	if = { limit = { has_global_flag = TFR_CRN_target_657 } 657 = { TFR_CRN_attach_tier_state = yes } }
	if = { limit = { has_global_flag = TFR_CRN_target_1328 } 1328 = { TFR_CRN_attach_tier_state = yes } }
	if = { limit = { has_global_flag = TFR_CRN_target_1293 } 1293 = { TFR_CRN_attach_tier_state = yes } }
	if = { limit = { has_global_flag = TFR_CRN_target_560 } 560 = { TFR_CRN_attach_tier_state = yes } }
	if = { limit = { has_global_flag = TFR_CRN_target_329 } 329 = { TFR_CRN_attach_tier_state = yes } }
}

##############################################################################
# 按档位挂载修正 — 运行在 STATE 作用域（当前目标州）
# 同档位同时挂 prc 系 + sov 系（数值一致，仅图/名不同）
##############################################################################
