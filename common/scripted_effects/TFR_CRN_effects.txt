# TFR_中俄谈判 - 核心脚本效果
# namespace: TFR_CRN

##############################################################################
# 谈判初始化 — 由调试决议触发
# 设置全局旗标 + SOV 初始变量
##############################################################################
TFR_CRN_init_negotiation = {
	set_global_flag = TFR_CRN_negotiation_active
	set_global_flag = TFR_CRN_negotiation_accessible
	PRC = { set_country_flag = TFR_CRN_PRC_started_negotiation }
	SOV = {
		set_variable = { var = TFR_CRN_desperation value = 0 }
		set_variable = { var = TFR_CRN_negotiation_days value = 0 }
	}
	news_event = { id = TFR_CRN.1 }
}

##############################################################################
# 计算 SOV 绝望度 — 每次 PRC 打开决议面板 / 点击购买前重算
# 运行在 SOV 作用域
##############################################################################
TFR_CRN_calc_desperation = {
	set_temp_variable = { CRN_desp = 0 }
	
	# SOV 在战争中
	if = { limit = { has_war = yes } add_to_temp_variable = { CRN_desp = 10 } }
	
	# 投降倾向（0~100% → 0~40）
	set_temp_variable = { CRN_surrender = surrender_progress }
	multiply_temp_variable = { CRN_surrender = 0.4 }
	round_variable = CRN_surrender
	add_to_temp_variable = { CRN_desp = CRN_surrender }
	
	# 战争伤亡对战争支持度的冲击（has_casualties_war_support 0~100 → 0~30）
	set_temp_variable = { CRN_casualties = has_casualties_war_support }
	multiply_temp_variable = { CRN_casualties = 0.3 }
	round_variable = CRN_casualties
	add_to_temp_variable = { CRN_desp = CRN_casualties }
	
	# 稳定度（低于 100% 开始扣，0% 扣 20）
	set_temp_variable = { CRN_stab = stability }
	multiply_temp_variable = { CRN_stab = 100 }
	subtract_from_temp_variable = { CRN_stab = 100 }   # 低于 100% 的部分
	if = { limit = { check_temp_variable = { CRN_stab < 0 } } set_temp_variable = { CRN_stab = 0 } }
	if = { limit = { check_temp_variable = { CRN_stab > 20 } } set_temp_variable = { CRN_stab = 20 } }
	round_variable = CRN_stab
	add_to_temp_variable = { CRN_desp = CRN_stab }
	
	# 谈判天数加成（每天 +0.5，最多 +50）
	add_to_variable = { var = TFR_CRN_negotiation_days value = 1 }
	set_temp_variable = { CRN_days = TFR_CRN_negotiation_days }
	multiply_temp_variable = { CRN_days = 0.5 }
	if = { limit = { check_temp_variable = { CRN_days > 50 } } set_temp_variable = { CRN_days = 50 } }
	round_variable = CRN_days
	add_to_temp_variable = { CRN_desp = CRN_days }
	
	# 钳位 0~100
	if = { limit = { check_temp_variable = { CRN_desp < 0 } } set_temp_variable = { CRN_desp = 0 } }
	if = { limit = { check_temp_variable = { CRN_desp > 100 } } set_temp_variable = { CRN_desp = 100 } }
	
	set_variable = { var = TFR_CRN_desperation value = CRN_desp }
}

##############################################################################
# 领土转交核心逻辑 — 调用前需确定购买的省 ID
# 调用前需在 PRC 作用域
# 参数：TFR_CRN_transfer_state_target（省ID）
##############################################################################
TFR_CRN_transfer_state = {
	# SOV 去核心
	SOV = { PREV = { remove_core_of = SOV } }
	# PRC 加核心 + 转移所有权
	PREV = {
		add_core_of = PRC
		set_controller = PRC
		set_owner = PRC
	}
}
