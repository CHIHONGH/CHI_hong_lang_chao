improve_people_democracy = {
    if = {
        limit = {
            has_idea = PRC_road_to_people_democracy
        }
        swap_ideas = {
            remove_idea = PRC_road_to_people_democracy
            add_idea = PRC_road_to_people_democracy1
        }
    }
    else_if = {
        limit = {
            has_idea = PRC_road_to_people_democracy1
        }
        swap_ideas = {
            remove_idea = PRC_road_to_people_democracy1
            add_idea = PRC_road_to_people_democracy2
        }
    }
    else_if = {
        limit = {
            has_idea = PRC_road_to_people_democracy2
        }
        swap_ideas = {
            remove_idea = PRC_road_to_people_democracy2
            add_idea = PRC_road_to_people_democracy3
        }

    else_if = {
        limit = {
            has_idea = PRC_road_to_people_democracy3
        }
        swap_ideas = {
            remove_idea = PRC_road_to_people_democracy3
            add_idea = PRC_road_to_people_democracy4
        }
    }
}
PLA_add_propaganda_effect = {
	if = {
		limit = {
			check_variable = {
				RD_PLA_political_slot_left_var > 1
			}
			check_variable = {
				RD_PLA_political_slot_middle_var > 1
			}
			check_variable = {
				RD_PLA_political_slot_right_var > 1
			}
		}
		log = "There is no propaganda slot. What are you doing?"
	}
	else_if = {
		limit = {
			check_variable = {
				RD_PLA_political_slot_left_var = 1
			}
		}
		set_variable = {
			RD_PLA_political_slot_left_var = RD_PLA_ppgd_temp
		}
		subtract_from_temp_variable = {
			RD_PLA_ppgd_temp = 1
		}
		meta_effect = {
			text = {
				add_timed_idea = {
					idea = RD_PLA_political_ppgd_[ppgd_var]
					days = 100
				}
			}
			ppgd_var = RD_PLA_ppgd_temp
		}
		set_country_flag = {
			flag = RD_PLA_political_slot_left_occupied_flag
			value = 1
			days = 100
		}
		subtract_from_variable = {
			RD_PLA_political_available_slots = 1
		}
	}
	else_if = {
		limit = {
			check_variable = {
				RD_PLA_political_slot_middle_var = 1
			}
		}
		set_variable = {
			RD_PLA_political_slot_middle_var = RD_PLA_ppgd_temp
		}
		subtract_from_temp_variable = {
			RD_PLA_ppgd_temp = 1
		}
		meta_effect = {
			text = {
				add_timed_idea = {
					idea = RD_PLA_political_ppgd_[ppgd_var]
					days = 100
				}
			}
			ppgd_var = RD_PLA_ppgd_temp
		}
		set_country_flag = {
			flag = RD_PLA_political_slot_middle_occupied_flag
			value = 1
			days = 100
		}
		subtract_from_variable = {
			RD_PLA_political_available_slots = 1
		}
	}
	else_if = {
		limit = {
			check_variable = {
				RD_PLA_political_slot_right_var = 1
			}
		}
		set_variable = {
			RD_PLA_political_slot_right_var = RD_PLA_ppgd_temp
		}
		subtract_from_temp_variable = {
			RD_PLA_ppgd_temp = 1
		}
		meta_effect = {
			text = {
				add_timed_idea = {
					idea = RD_PLA_political_ppgd_[ppgd_var]
					days = 100
				}
			}
			ppgd_var = RD_PLA_ppgd_temp
		}
		set_country_flag = {
			flag = RD_PLA_political_slot_right_occupied_flag
			value = 1
			days = 100
		}
		subtract_from_variable = {
			RD_PLA_political_available_slots = 1
		}
	}
}

PLA_stop_propaganda_effect = {
	if = {
		limit = {
			check_variable = {
				RD_PLA_political_slot_left_var = RD_PLA_ppgd_temp
			}
		}
		set_variable = {
			RD_PLA_political_slot_left_var = 1
		}
		add_to_variable = {
			RD_PLA_political_available_slots = 1
		}
	}
	else_if = {
		limit = {
			check_variable = {
				RD_PLA_political_slot_middle_var = RD_PLA_ppgd_temp
			}
		}
		set_variable = {
			RD_PLA_political_slot_middle_var = 1
		}
		add_to_variable = {
			RD_PLA_political_available_slots = 1
		}
	}
	else_if = {
		limit = {
			check_variable = {
				RD_PLA_political_slot_right_var = RD_PLA_ppgd_temp
			}
		}
		set_variable = {
			RD_PLA_political_slot_right_var = 1
		}
		add_to_variable = {
			RD_PLA_political_available_slots = 1
		}
	}
	else = {
		log = "Is there a propaganda that I should remove? Maybe not."
	}
}
