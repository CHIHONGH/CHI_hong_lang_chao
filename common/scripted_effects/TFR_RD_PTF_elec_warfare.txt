# 削弱 1 次（次数 +1 后按新档位置档），上限 10 次
PTF_elec_weaken = {
	PTF = {
		add_to_variable = { PTF_elec_count = 1 }
		if = {
			limit = {
				NOT = { check_variable = { PTF_elec_count < 10 } }
			}
			set_variable = { PTF_elec_count = 10 }
		}
	}
	PTF_elec_set_stage = yes
}

# 体系彻底崩溃（占领西雅图，直接置终档）
PTF_elec_collapse = {
	PTF = {
		set_variable = { PTF_elec_count = 10 }
	}
	PTF_elec_set_stage = yes
}
