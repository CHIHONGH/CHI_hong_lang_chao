# TFR_中俄谈判 — 动态脚本本地化
# 根据 SOV 意识形态和领导人动态切换"苏方/俄方"称谓

defined_text = {
	name = TFR_CRN_GetSovFactionLabel

	# 泛红且非杜金欧亚 → 苏方
	text = {
		trigger = {
			OR = {
				has_government = libertarian_socialist
				has_government = communism
				has_government = totalitarian_socialist
			}
			NOT = {
				has_country_leader = {
					character = SOV_alexander_dugin
					ruling_only = yes
				}
			}
		}
		localization_key = TFR_CRN_label_soviet
	}

	# 其余 → 俄方
	text = {
		localization_key = TFR_CRN_label_russian
	}
}
