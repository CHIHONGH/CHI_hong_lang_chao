# ==================== PTF 难度激活（难度选择已移至游戏规则，本事件由规则侧接线触发） ====================
country_event = {
	id = red_dawn_guide.7
	hidden = yes
	fire_only_once = yes

	trigger = {
		tag = PTF
		NOT = {
			OR = {
				has_global_flag = USA_civil_war_project_winner_USB_caligula_flag
				has_global_flag = USA_civil_war_project_winner_USB_NSS_flag
			}
		}
	}

	immediate = {
		if = {
			limit = { has_game_rule = { rule = The_PTF_Difficulty_Choose option = PTF_Baby_Mode } }
			add_ideas = {
				PTF_Fascist_military_revolution_II_idea
				PTF_Counter_revolutionary_center_II_idea
			}
		}
		if = {
			limit = { has_game_rule = { rule = The_PTF_Difficulty_Choose option = PTF_Simple_Mode } }
			add_ideas = {
				PTF_Fascist_military_revolution_I_idea
				PTF_Counter_revolutionary_center_I_idea
			}
		}
		if = {
			limit = { has_game_rule = { rule = The_PTF_Difficulty_Choose option = PTF_Normal_Mode } }
			add_ideas = {
				PTF_Fascist_military_revolution_I_idea
				PTF_Counter_revolutionary_center_I_idea
				PTF_Anti_communist_fortress_idea
			}
		}
		if = {
			limit = { has_game_rule = { rule = The_PTF_Difficulty_Choose option = PTF_Hard_Mode } }
			add_ideas = {
				PTF_Fascist_military_revolution_II_idea
				PTF_Counter_revolutionary_center_II_idea
				PTF_Anti_communist_fortress_idea
			}
			set_technology = {
				base_cruise_antiship_missile = 1
				improved_cruise_antiship_missile = 1
				ASBM_launcher = 1
				ship_super_deck = 1
			}
		}
		if = {
			limit = { has_game_rule = { rule = The_PTF_Difficulty_Choose option = PTF_Extremely_Hard_Mode } }
			add_ideas = {
				PTF_Fascist_military_revolution_III_idea
				PTF_Counter_revolutionary_center_III_idea
				PTF_Anti_communist_fortress_idea
				PTF_End_of_communism_idea
			}
			set_technology = {
				base_cruise_antiship_missile = 1
				improved_cruise_antiship_missile = 1
				ASBM_launcher = 1
				ship_super_deck = 1
			}
		}
		set_technology = { tide_of_iron = 1 }
	}
}
