# ==================== PTF 难度激活（保留原有逻辑） ====================
country_event = {
	id = pdto_difficulty_choose.7
	hidden = yes
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		tag = PTF
	}

	immediate = {
		if = {
			limit = { PRC = { has_country_flag = PTF_difficulty_Simple } }
			add_ideas = {
				PTF_Fascist_military_revolution_I_idea
				PTF_Counter_revolutionary_center_I_idea
			}
		}
		if = {
			limit = { PRC = { has_country_flag = PTF_difficulty_Normal } }
			add_ideas = {
				PTF_Fascist_military_revolution_II_idea
				PTF_Counter_revolutionary_center_II_idea
				PTF_Anti_communist_fortress_idea
			}
		}
		if = {
			limit = { PRC = { has_country_flag = PTF_difficulty_Hard } }
			add_ideas = {
				PTF_Fascist_military_revolution_III_idea
				PTF_Counter_revolutionary_center_III_idea
				PTF_Anti_communist_fortress_idea
			}
		}
		if = {
			limit = { PRC = { has_country_flag = PTF_difficulty_Extremely_hard } }
			add_ideas = {
				PTF_Fascist_military_revolution_III_idea
				PTF_Counter_revolutionary_center_III_idea
				PTF_Anti_communist_fortress_idea
				PTF_End_of_communism_idea
			}
		}
		set_technology = { tide_of_iron = 1 }
	}
}
