# ==================== PTF 难度激活（难度选择已移至游戏规则，本事件由规则侧接线触发） ====================
country_event = {
	id = red_dawn_guide.7
	hidden = yes
	fire_only_once = yes

	trigger = {
		tag = PTF
	}

	immediate = {
		if = {
			limit = { PRC = { has_game_rule = { rule = The_PTF_Difficulty_Choose option = PTF_Simple_Mode } } }
			add_ideas = {
				PTF_Fascist_military_revolution_I_idea
				PTF_Counter_revolutionary_center_I_idea
			}
		}
		if = {
			limit = { PRC = { has_game_rule = { rule = The_PTF_Difficulty_Choose option = PTF_Normal_Mode } } }
			add_ideas = {
				PTF_Fascist_military_revolution_II_idea
				PTF_Counter_revolutionary_center_II_idea
				PTF_Anti_communist_fortress_idea
			}
		}
		if = {
			limit = { PRC = { has_game_rule = { rule = The_PTF_Difficulty_Choose option = PTF_Hard_Mode } } }
			add_ideas = {
				PTF_Fascist_military_revolution_III_idea
				PTF_Counter_revolutionary_center_III_idea
				PTF_Anti_communist_fortress_idea
			}
		}
		if = {
			limit = { PRC = { has_game_rule = { rule = The_PTF_Difficulty_Choose option = PTF_Extremely_Hard_Mode } } }
			add_ideas = {
				PTF_Fascist_military_revolution_III_idea
				PTF_Counter_revolutionary_center_III_idea
				PTF_Anti_communist_fortress_idea
				PTF_End_of_communism_idea
			}
		}
		set_technology = { tide_of_iron = 1 }
	}
}
