# ==================== 南海对峙结束后的获知事件 ====================
# 挂载：prcs.3146 主选项 hidden_effect 预排（days=5），仅一次
# 效果：PTF 初始化幽灵屏障变量并挂动态精神
country_event = {
	id = prc_elec_warfare.101
	title = prc_elec_warfare.101.t
	desc = prc_elec_warfare.101.d
	picture = GFX_Event_USA_fleet
	is_triggered_only = yes

	immediate = {
		# 动态修正挂载与初始化分离：修正无条件挂载（has_dynamic_modifier 守卫防重挂，
		# 避免南海获胜削弱先行时 count>0 导致整块跳过、修正永不挂载）
		PTF = {
			if = {
				limit = {
					NOT = { has_dynamic_modifier = { modifier = PTF_elec_warfare_dynamic } }
				}
				add_dynamic_modifier = { modifier = PTF_elec_warfare_dynamic }
			}
		}
		# 仅当尚未发生任何削弱时初始化（南海获胜削弱与获知预排同天发生，
		# 若 count 已被削弱置 1，跳过硬初始化防止覆盖档位）；
		# 数值以档位表为准，直接调用 set_stage 置 count=0 档（vis -0.85/pos -0.25），
		# 与 PTF_elec_set_stage、假精神 0 档一致，避免手写数值漂移
		if = {
			limit = {
				PTF = {
					NOT = { check_variable = { PTF_elec_count > 0 } }
				}
			}
			PTF_elec_set_stage = yes
		}
	}

	option = {
		name = prc_elec_warfare.101.a
		effect_tooltip = {
			PTF_elec_dummy_stage = yes
		}
	}
}
